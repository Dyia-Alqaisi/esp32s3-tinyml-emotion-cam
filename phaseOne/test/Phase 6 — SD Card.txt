#include <Arduino.h>
#include "FS.h"
#include "SD_MMC.h"

// =========================================================
// Freenove built-in microSD slot
// =========================================================

constexpr int SD_PIN_CMD = 38;
constexpr int SD_PIN_CLK = 39;
constexpr int SD_PIN_D0  = 40;

constexpr const char* SD_MOUNT_POINT = "/sdcard";

constexpr const char* TEXT_TEST_PATH =
    "/logs/phase3_test.txt";

constexpr const char* BINARY_TEST_PATH =
    "/models/io_test.bin";

constexpr size_t BINARY_TEST_SIZE = 1024 * 1024; // 1 MB
constexpr size_t BLOCK_SIZE = 4096;

uint8_t ioBuffer[BLOCK_SIZE];

// =========================================================
// Utility functions
// =========================================================

void stopWithError(const char* message)
{
    Serial.println();
    Serial.println("========================================");
    Serial.println("PHASE 3 RESULT: FAILED");
    Serial.println(message);
    Serial.println("========================================");

    while (true) {
        delay(1000);
    }
}

const char* getCardTypeName(uint8_t cardType)
{
    switch (cardType) {
        case CARD_MMC:
            return "MMC";

        case CARD_SD:
            return "SDSC";

        case CARD_SDHC:
            return "SDHC/SDXC";

        case CARD_NONE:
            return "No card";

        default:
            return "Unknown";
    }
}

bool ensureDirectory(const char* path)
{
    if (SD_MMC.exists(path)) {
        Serial.printf("Directory exists : %s\n", path);
        return true;
    }

    Serial.printf("Creating directory: %s\n", path);

    if (!SD_MMC.mkdir(path)) {
        Serial.printf("Failed to create : %s\n", path);
        return false;
    }

    return true;
}

// =========================================================
// Text-file test
// =========================================================

bool runTextFileTest()
{
    constexpr const char* expectedText =
        "ALQAI EmotionCam Phase 3 SD test passed";

    if (SD_MMC.exists(TEXT_TEST_PATH)) {
        SD_MMC.remove(TEXT_TEST_PATH);
    }

    Serial.printf("Writing text file: %s\n", TEXT_TEST_PATH);

    File file = SD_MMC.open(TEXT_TEST_PATH, FILE_WRITE);

    if (!file) {
        Serial.println("Could not open text file for writing.");
        return false;
    }

    const size_t written = file.println(expectedText);

    file.flush();
    file.close();

    if (written == 0) {
        Serial.println("Nothing was written to the text file.");
        return false;
    }

    Serial.printf("Reading text file: %s\n", TEXT_TEST_PATH);

    file = SD_MMC.open(TEXT_TEST_PATH, FILE_READ);

    if (!file) {
        Serial.println("Could not open text file for reading.");
        return false;
    }

    String actualText = file.readStringUntil('\n');

    file.close();

    actualText.trim();

    Serial.printf("Read content     : %s\n", actualText.c_str());

    if (actualText != expectedText) {
        Serial.println("Text verification failed.");
        return false;
    }

    Serial.println("Text verification: PASS");

    return true;
}

// =========================================================
// 1 MB binary-file test
// =========================================================

bool runBinaryFileTest()
{
    if (SD_MMC.exists(BINARY_TEST_PATH)) {
        SD_MMC.remove(BINARY_TEST_PATH);
    }

    Serial.printf(
        "Writing binary test: %u bytes\n",
        static_cast<unsigned int>(BINARY_TEST_SIZE)
    );

    File file = SD_MMC.open(BINARY_TEST_PATH, FILE_WRITE);

    if (!file) {
        Serial.println("Could not open binary file for writing.");
        return false;
    }

    const uint32_t writeStart = millis();

    size_t totalWritten = 0;
    size_t blockIndex = 0;

    while (totalWritten < BINARY_TEST_SIZE) {
        const size_t remaining =
            BINARY_TEST_SIZE - totalWritten;

        const size_t bytesThisBlock =
            remaining < BLOCK_SIZE ? remaining : BLOCK_SIZE;

        /*
         * Generate a predictable pattern so the read test
         * can verify that every byte is correct.
         */
        for (size_t i = 0; i < bytesThisBlock; ++i) {
            ioBuffer[i] =
                static_cast<uint8_t>((blockIndex + i) & 0xFF);
        }

        const size_t written =
            file.write(ioBuffer, bytesThisBlock);

        if (written != bytesThisBlock) {
            Serial.printf(
                "Binary write failed at byte %u\n",
                static_cast<unsigned int>(totalWritten)
            );

            file.close();
            return false;
        }

        totalWritten += written;
        blockIndex++;
    }

    file.flush();
    file.close();

    const uint32_t writeDuration =
        max<uint32_t>(millis() - writeStart, 1);

    const float writeSpeedKBps =
        (static_cast<float>(totalWritten) / 1024.0f) /
        (static_cast<float>(writeDuration) / 1000.0f);

    Serial.printf(
        "Binary write     : PASS — %.1f KB/s\n",
        writeSpeedKBps
    );

    // -----------------------------------------------------
    // Read and verify the complete binary file
    // -----------------------------------------------------

    file = SD_MMC.open(BINARY_TEST_PATH, FILE_READ);

    if (!file) {
        Serial.println("Could not open binary file for reading.");
        return false;
    }

    if (file.size() != BINARY_TEST_SIZE) {
        Serial.printf(
            "Wrong binary size: expected %u, found %u\n",
            static_cast<unsigned int>(BINARY_TEST_SIZE),
            static_cast<unsigned int>(file.size())
        );

        file.close();
        return false;
    }

    const uint32_t readStart = millis();

    size_t totalRead = 0;
    blockIndex = 0;

    while (totalRead < BINARY_TEST_SIZE) {
        const size_t remaining =
            BINARY_TEST_SIZE - totalRead;

        const size_t bytesThisBlock =
            remaining < BLOCK_SIZE ? remaining : BLOCK_SIZE;

        const size_t bytesRead =
            file.read(ioBuffer, bytesThisBlock);

        if (bytesRead != bytesThisBlock) {
            Serial.printf(
                "Binary read failed at byte %u\n",
                static_cast<unsigned int>(totalRead)
            );

            file.close();
            return false;
        }

        for (size_t i = 0; i < bytesRead; ++i) {
            const uint8_t expectedByte =
                static_cast<uint8_t>((blockIndex + i) & 0xFF);

            if (ioBuffer[i] != expectedByte) {
                Serial.printf(
                    "Data mismatch at byte %u\n",
                    static_cast<unsigned int>(totalRead + i)
                );

                file.close();
                return false;
            }
        }

        totalRead += bytesRead;
        blockIndex++;
    }

    file.close();

    const uint32_t readDuration =
        max<uint32_t>(millis() - readStart, 1);

    const float readSpeedKBps =
        (static_cast<float>(totalRead) / 1024.0f) /
        (static_cast<float>(readDuration) / 1000.0f);

    Serial.printf(
        "Binary read      : PASS — %.1f KB/s\n",
        readSpeedKBps
    );

    Serial.println("Binary integrity : PASS");

    // Remove only the temporary binary test file.
    if (!SD_MMC.remove(BINARY_TEST_PATH)) {
        Serial.println(
            "Warning: could not delete temporary binary file."
        );
    } else {
        Serial.println("Temporary binary file removed.");
    }

    return true;
}

// =========================================================
// Arduino setup
// =========================================================

void setup()
{
    Serial.begin(115200);

    // Give native USB CDC time to reconnect.
    delay(2500);

    Serial.println();
    Serial.println("========================================");
    Serial.println(" ALQAI EmotionCam - SD Validation");
    Serial.println("========================================");

    Serial.printf("SD CMD          : GPIO%d\n", SD_PIN_CMD);
    Serial.printf("SD CLK          : GPIO%d\n", SD_PIN_CLK);
    Serial.printf("SD D0           : GPIO%d\n", SD_PIN_D0);
    Serial.println("SD mode         : SDMMC 1-bit");
    Serial.println("SD frequency    : 20 MHz default");

    /*
     * setPins order:
     * CLK, CMD, D0
     */
    if (!SD_MMC.setPins(
            SD_PIN_CLK,
            SD_PIN_CMD,
            SD_PIN_D0
        )) {
        stopWithError("SD_MMC.setPins() failed.");
    }

    /*
     * Parameters:
     * 1. Mount point
     * 2. true  = use one-bit mode
     * 3. false = do not format automatically
     * 4. Default SDMMC frequency
     * 5. Maximum number of open files
     */
    const bool mounted = SD_MMC.begin(
        SD_MOUNT_POINT,
        true,
        false,
        SDMMC_FREQ_DEFAULT,
        5
    );

    if (!mounted) {
        stopWithError(
            "Card mount failed. Check FAT32 format and insertion."
        );
    }

    const uint8_t cardType = SD_MMC.cardType();

    if (cardType == CARD_NONE) {
        stopWithError("No microSD card detected.");
    }

    Serial.println("----------------------------------------");
    Serial.println("SD mount        : PASS");
    Serial.printf(
        "Card type       : %s\n",
        getCardTypeName(cardType)
    );

    Serial.printf(
        "Card capacity   : %llu MB\n",
        SD_MMC.cardSize() / (1024ULL * 1024ULL)
    );

    Serial.printf(
        "Filesystem size : %llu MB\n",
        SD_MMC.totalBytes() / (1024ULL * 1024ULL)
    );

    Serial.printf(
        "Used space      : %llu MB\n",
        SD_MMC.usedBytes() / (1024ULL * 1024ULL)
    );

    Serial.println("----------------------------------------");

    if (!ensureDirectory("/models")) {
        stopWithError("Could not create /models.");
    }

    if (!ensureDirectory("/captures")) {
        stopWithError("Could not create /captures.");
    }

    if (!ensureDirectory("/config")) {
        stopWithError("Could not create /config.");
    }

    if (!ensureDirectory("/logs")) {
        stopWithError("Could not create /logs.");
    }

    Serial.println("----------------------------------------");

    if (!runTextFileTest()) {
        stopWithError("Text-file test failed.");
    }

    Serial.println("----------------------------------------");

    if (!runBinaryFileTest()) {
        stopWithError("Binary-file test failed.");
    }

    Serial.println("----------------------------------------");
    Serial.println("PHASE 3 RESULT  : ALL TESTS PASSED");
    Serial.println("SD card is ready for later model storage.");
    Serial.println("========================================");
}

// =========================================================
// Arduino loop
// =========================================================

void loop()
{
    static uint32_t lastHeartbeat = 0;

    if (millis() - lastHeartbeat >= 5000) {
        lastHeartbeat = millis();

        Serial.printf(
            "[SD heartbeat] uptime=%lu s | heap=%u | PSRAM=%u\n",
            millis() / 1000,
            ESP.getFreeHeap(),
            ESP.getFreePsram()
        );
    }

    delay(10);
}
