[env:4d_systems_esp32s3_gen4_r8n16]
platform = espressif32
board = 4d_systems_esp32s3_gen4_r8n16
framework = arduino

; =========================================================
; Freenove ESP32-S3-WROOM N16R8
; 16 MB Flash + 8 MB OPI PSRAM
; =========================================================
board_build.flash_size = 16MB
board_upload.flash_size = 16MB

; Two 6.25 MB application partitions + SPIFFS
board_build.partitions = default_16MB.csv
board_upload.maximum_size = 6553600

; =========================================================
; Native physical USB/OTG connector
; Hardware USB CDC/JTAG — NOT UART
; =========================================================

; Remove the Freenove profile's CDC-disabled defaults
build_unflags =
    -DARDUINO_USB_CDC_ON_BOOT=0
    -DARDUINO_USB_MODE=0

; Enable Hardware USB CDC/JTAG
build_flags =
    -DARDUINO_USB_CDC_ON_BOOT=1
    -DARDUINO_USB_MODE=1

; Upload through the native USB connector
upload_protocol = esptool
upload_speed = 115200

; Handle USB-port reset and re-enumeration during upload
board_upload.use_1200bps_touch = true
board_upload.wait_for_upload_port = true
board_upload.require_upload_port = true

; =========================================================
; Serial Monitor through native USB CDC
; =========================================================
monitor_speed = 115200



lib_deps =
    adafruit/Adafruit GFX Library @ ^1.12.1
    adafruit/Adafruit SSD1306 @ ^2.5.15
